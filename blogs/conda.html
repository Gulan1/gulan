    Conda常用指令

    Gulan 5/20/2025 PythonConda

    # Conda常用指令

    # 环境管理

    # 创建环境

    # 创建一个指定Python版本的环境
conda create -n myenv python=3.8

# 有yaml配置文件创建环境
conda myenv create -f environment.yaml

# 更新环境
conda myenv update -f environment.yaml --prune

    # 激活/切换环境

    # 激活环境
conda activate myenv

# 退出当前环境
conda deactivate

    # 环境信息

    # 列出所有环境
conda env list
conda info --envs

# 查看当前环境中安装的包
conda list

    # 删除环境

    # 删除指定环境
conda remove -n myenv --all

# 清楚缓存
conda clean --all

    # 复制环境

    conda create -n newenv --clone myenv

    # 包管理

    # 安装包

    # 在当前环境中安装包
conda install pandas

# 指定版本安装
conda install pandas=1.0.0

# 从特定通道安装
conda install -c conda-forge matplotlib

    # 更新包

    # 更新单个包
conda update pandas

# 更新所有包
conda update --all

# 更新conda自身
conda update conda

    # 删除包

    conda remove pandas

    # 导出环境配置

    # 导出环境配置到YAML文件
conda env export > environment.yml

    # 搜索包

    conda search pandas

    # Conda自身管理

    # 查看Conda信息

    conda info

    # 配置通道

    # 添加通道
conda config --add channels conda-forge

# 查看已配置的通道
conda config --show channels

    # 常见问题解决

    # 环境损坏修复

    conda update --force conda

    # 包冲突解决

    # 使用严格通道优先级
conda config --set channel_priority strict

    # Miniconda安装

    # 下载Miniconda安装脚本
wget https://repo.anaconda.com/miniconda/Miniconda3-latest-Linux-x86_64.sh

# 添加执行权限
sudo chmod +x Miniconda3-latest-Linux-x86_64.sh

# 运行安装脚本
./Miniconda3-latest-Linux-x86_64.sh

# 添加到环境变量
export PATH="/home/lvl/miniconda3/bin:$PATH"

    # 其他配置

    # 禁用自动激活base环境
conda config --set auto_activate_base false

    # 配置国内镜像源

    # 添加清华镜像源
conda config --add channels https://mirrors.tuna.tsinghua.edu.cn/anaconda/pkgs/free/
conda config --add channels https://mirrors.tuna.tsinghua.edu.cn/anaconda/pkgs/main/
conda config --set show_channel_urls yes

    # 还原脚本

    # 还原为默认配置
conda config --remove-key channels
conda config --set show_channel_urls no
